from datetime import timedelta from django.conf import settings from django.core.management.base import BaseCommand from django.core.mail import send_mail from django.utils import timezone from assets.models import Asset class Command(BaseCommand): help = "Send maintenance and warranty reminders for assets due soon." def handle(self, *args, **options): today = timezone.now().date() due_assets = Asset.objects.filter( maintenance_schedule__next_maintenance_date__isnull=False, maintenance_schedule__next_maintenance_date__lte=today + timedelta(days=30), ).select_related("maintenance_schedule") sent = 0 for asset in due_assets: if not settings.DEFAULT_FROM_EMAIL: continue subject = f"Asset maintenance reminder: {asset.asset_tag}" message = ( f"Asset {asset.name} ({asset.asset_tag}) is due for maintenance on " f"{asset.maintenance_schedule.next_maintenance_date}. " f"Current status: {asset.get_status_display()}." ) send_mail(subject, message, settings.DEFAULT_FROM_EMAIL, [settings.DEFAULT_FROM_EMAIL], fail_silently=True) sent += 1 warranty_assets = Asset.objects.filter( warranty_end__isnull=False, warranty_end__lte=today + timedelta(days=90), ) for asset in warranty_assets: subject = f"Warranty expiration reminder: {asset.asset_tag}" message = ( f"The warranty for {asset.name} ({asset.asset_tag}) expires on {asset.warranty_end}. " f"Please review service plans and maintenance scheduling." ) send_mail(subject, message, settings.DEFAULT_FROM_EMAIL, [settings.DEFAULT_FROM_EMAIL], fail_silently=True) sent += 1 self.stdout.write(self.style.SUCCESS(f"Maintenance reminders processed: {sent} notifications."))
